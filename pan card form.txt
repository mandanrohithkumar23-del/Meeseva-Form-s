<script>
    // Auto-advance focus to next box on typing
    document.addEventListener('input', function (e) {
        if (e.target.classList.contains('char-input')) {
            if (e.target.value.length >= e.target.maxLength) {
                let next = e.target.nextElementSibling;
                while (next) {
                    if (next.classList.contains('char-input')) {
                        next.focus();
                        break;
                    }
                    next = next.nextElementSibling;
                }
            }
        }
    });

    // Move back to previous box on backspace
    document.addEventListener('keydown', function (e) {
        if (e.target.classList.contains('char-input') && e.key === 'Backspace' && e.target.value === '') {
            let prev = e.target.previousElementSibling;
            while (prev) {
                if (prev.classList.contains('char-input')) {
                    prev.focus();
                    break;
                }
                prev = prev.previousElementSibling;
            }
        }
    });
</script>
