    <script>
        // Generate character boxes dynamically
        document.querySelectorAll('.char-boxes').forEach(container => {
            const length = parseInt(container.getAttribute('data-length'));
            const defaultValue = container.getAttribute('data-default') || '';
            
            for (let i = 0; i < length; i++) {
                const input = document.createElement('input');
                input.type = 'text';
                input.maxLength = 1;
                input.className = 'char-box';
                if (defaultValue[i]) {
                    input.value = defaultValue[i];
                }
                
                // Auto Tab functionality
                input.addEventListener('input', (e) => {
                    if (e.target.value.length === 1) {
                        const next = e.target.nextElementSibling;
                        if (next && next.classList.contains('char-box')) {
                            next.focus();
                        } else {
                            // Try finding next box in next group if any
                            const parent = e.target.parentElement;
                            if (parent.nextElementSibling && parent.nextElementSibling.classList.contains('char-boxes')) {
                                const firstBox = parent.nextElementSibling.querySelector('.char-box');
                                if (firstBox) firstBox.focus();
                            }
                        }
                    }
                });

                input.addEventListener('keydown', (e) => {
                    if (e.key === 'Backspace' && e.target.value === '') {
                        const prev = e.target.previousElementSibling;
                        if (prev && prev.classList.contains('char-box')) {
                            prev.focus();
                        } else {
                            const parent = e.target.parentElement;
                            if (parent.previousElementSibling && parent.previousElementSibling.classList.contains('char-boxes')) {
                                const boxes = parent.previousElementSibling.querySelectorAll('.char-box');
                                if (boxes.length) boxes[boxes.length - 1].focus();
                            }
                        }
                    }
                });

                container.appendChild(input);
            }
        });

        let currentLang = 'en';
        function toggleLanguage() {
            const enElements = document.querySelectorAll('.lang-en');
            const teElements = document.querySelectorAll('.lang-te');

            if (currentLang === 'en') {
                enElements.forEach(el => el.style.display = 'none');
                teElements.forEach(el => el.style.display = 'inline');
                currentLang = 'te';
            } else {
                teElements.forEach(el => el.style.display = 'none');
                enElements.forEach(el => el.style.display = 'inline');
                currentLang = 'en';
            }
        }
    </script>
